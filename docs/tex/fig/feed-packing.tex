% Lane 0's eight accumulator slots, read by the next MMA as its B operand, under
% the two packing conventions.
\begin{figure}[htbp]
\centering
\begin{tikzpicture}[x=1mm, y=1mm,
    cell/.style={draw=ink!45, line width=0.4pt, minimum width=12mm, minimum height=6mm,
      inner sep=0pt, font=\scriptsize},
    rl/.style={anchor=east, font=\scriptsize, text=ink, align=right},
    hd/.style={font=\scriptsize, text=ink!60, anchor=south}]

  \def\w{12.6}
  \foreach \j in {0,...,7} \node[hd] at ({\j*\w+6},4) {slot \j};

  % Row 1: what the hardware writes, in canonical D coordinates.
  \node[rl] at (-2,0) {Canonical D element};
  \foreach \j in {0,...,3} \node[cell] at ({\j*\w+6},0) {(0,\,\j)};
  \foreach \j/\c in {4/0, 5/1, 6/2, 7/3} \node[cell] at ({\j*\w+6},0) {(1,\,\c)};

  % Row 2: what the next MMA reads out of the same registers, as B.
  \node[rl] at (-2,-8) {Read by the next MMA as B};
  \foreach \j in {0,...,3} \node[cell] at ({\j*\w+6},-8) {(0,\,\j)};
  \foreach \j/\c in {4/0, 5/1, 6/2, 7/3} \node[cell] at ({\j*\w+6},-8) {(8,\,\c)};

  % Row 3: canonical packing.
  \node[rl] at (-2,-17) {Canonical packing puts D row};
  \foreach \j in {0,...,3} \node[cell, fill=apple!14] at ({\j*\w+6},-17) {0};
  \foreach \j in {4,...,7} \node[cell, fill=apple!14] at ({\j*\w+6},-17) {1};

  % Row 4: this compiler's B-row packing.
  \node[rl] at (-2,-25) {B-row packing puts D row};
  \foreach \j in {0,...,3} \node[cell, fill=ours!16] at ({\j*\w+6},-25) {0};
  \foreach \j in {4,...,7} \node[cell, fill=ours!16] at ({\j*\w+6},-25) {8};

  % Slot 4 is the one that shows the difference.
  \draw[native, line width=0.9pt, rounded corners=1.5pt]
    ({4*\w-0.3},3) rectangle ({5*\w-0.3},-28.5);

  \node[font=\scriptsize, anchor=north west, text width=100mm, align=left] at (-2,-33)
    {\textbf{Slot 4} is where the two conventions part. Canonical packing puts canonical D row
     1 there, so B row 8 receives a rotated operand and the other operand's K index must undo
     it. B-row packing puts application row 8 there, and the consumer reads it directly.};
\end{tikzpicture}
\caption{One register, two meanings. Lane 0's eight accumulator slots, as the hardware writes them and as the next
MMA reads them when D is fed as B. Because \code{pos\_b(r, c) = pos(rotl1(r), c)}, canonical packing delivers D row
\code{rotl1(k)} where the consumer expects row k; this compiler's B-row packing puts application row k there instead
(\cref{tab:coordinate-example}).}
\label{fig:feed-packing}
\end{figure}
